cmake_minimum_required(VERSION 3.24)
project(strata_safetensors_source LANGUAGES CXX)

# Standalone opt-in tools: the engine's build, backends and defaults are untouched.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC AND CMAKE_CXX_CL_SHOWINCLUDES_PREFIX MATCHES "娉ㄦ剰")
    # CMake can double-decode the Chinese-only MSVC include prefix as GBK.
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件: ")
endif()

set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
add_library(strata_safetensors_source STATIC ${ROOT}/src/weights/safetensors.cpp)
target_include_directories(strata_safetensors_source PUBLIC ${ROOT}/include PRIVATE ${ROOT}/third_party)
add_executable(strata-safetensors-source ${ROOT}/src/weights/source_inspect.cpp)
target_link_libraries(strata-safetensors-source PRIVATE strata_safetensors_source)
target_include_directories(strata-safetensors-source PRIVATE ${ROOT}/third_party)

include(CTest)
set(SOURCE_TARGETS strata_safetensors_source strata-safetensors-source)
if(BUILD_TESTING)
    add_executable(safetensors_source_test ${ROOT}/src/weights/source_test.cpp)
    target_link_libraries(safetensors_source_test PRIVATE strata_safetensors_source)
    target_include_directories(safetensors_source_test PRIVATE ${ROOT}/third_party)
    list(APPEND SOURCE_TARGETS safetensors_source_test)
    add_test(NAME safetensors_source COMMAND safetensors_source_test)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME safetensors_reader COMMAND ${Python3_EXECUTABLE}
        ${ROOT}/tools/test_safetensors_reader.py --exe $<TARGET_FILE:strata-safetensors-source> -v)
endif()
foreach(target IN LISTS SOURCE_TARGETS)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /utf-8 /permissive-)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
endforeach()
