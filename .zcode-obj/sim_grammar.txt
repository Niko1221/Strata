answer-age-kv ::= "\"age\"" answer-space ":" answer-space answer-integer
answer-char ::= [^"\\\x7F\x00-\x1F] | [\\] (["\\bfnrt] | "u" [0-9a-fA-F]{4})
answer-city-kv ::= "\"city\"" answer-space ":" answer-space answer-string
answer-integer ::= ("-"? answer-integral-part)
answer-integral-part ::= [0] | [1-9] [0-9]{0,15}
answer-name-kv ::= "\"name\"" answer-space ":" answer-space answer-string
answer-root ::= "{" answer-space answer-name-kv "," answer-space answer-age-kv "," answer-space answer-city-kv answer-space "}"
answer-space ::= | " " | "\n"{1,2} [ \t]{0,20}
answer-string ::= "\"" answer-char* "\""
reason ::= [^<] reason | "<" reason-1 | ""
reason-1 ::= [^/] reason | "/" reason-2 | ""
reason-2 ::= [^t] reason | "t" reason-3 | ""
reason-3 ::= [^h] reason | "h" reason-4 | ""
reason-4 ::= [^i] reason | "i" reason-5 | ""
reason-5 ::= [^n] reason | "n" reason-6 | ""
reason-6 ::= [^k] reason | "k" reason-7 | ""
reason-7 ::= [^>] reason | ""
ws ::= " " | "	" | "\n"
answer ::= ws answer-root
root ::= reason "</think>\n" answer
