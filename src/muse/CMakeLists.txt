# Deliberately independent of strata_core and the Qwen numerical libraries.
if(NOT TARGET ggml-base)
  message(FATAL_ERROR "Muse requires the pinned GGML row converters (STRATA_NATIVE_EXPERTS=ON)")
endif()
add_library(strata_muse STATIC model.cpp)
target_link_libraries(strata_muse PUBLIC strata_artifact ggml-base)
target_compile_features(strata_muse PUBLIC cxx_std_17)
# gfx906 uses CUDA-named HIP aliases, without the NVIDIA toolkit or cuBLAS GEMV.
if(STRATA_ENABLE_CUDA AND NOT STRATA_HIP_GFX906)
  find_package(CUDAToolkit REQUIRED)
  target_sources(strata_muse PRIVATE cuda_matvec.cpp)
  target_compile_definitions(strata_muse PRIVATE STRATA_MUSE_CUDA=1)
  target_link_libraries(strata_muse PRIVATE CUDA::cudart CUDA::cublas)
endif()
add_executable(strata-muse main.cpp)
strata_utf8_manifest(strata-muse)
target_link_libraries(strata-muse PRIVATE strata_muse)
if(STRATA_BUILD_TESTS)
  add_executable(muse_model_test model_test.cpp test_allocations.cpp)
  strata_utf8_manifest(muse_model_test)
  target_link_libraries(muse_model_test PRIVATE strata_muse)
  add_test(NAME muse_model_test COMMAND muse_model_test)
endif()
