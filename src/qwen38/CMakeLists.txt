cmake_minimum_required(VERSION 3.25)
project(strata_q38_owned LANGUAGES C CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 120)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
find_package(CUDAToolkit 13 REQUIRED)
option(STRATA_Q38_REQUIRE_CUDA135 "Require CUDA 13.5 compiler and Toolkit for the candidate" OFF)
option(STRATA_Q38_EXPERIMENTAL_FUSED_WMMA "Enable unqualified packed-weight WMMA prefill candidate" OFF)
option(STRATA_Q38_FASTMATH "Diagnostic: compile with --use_fast_math (approximate exp/div/sqrt)" OFF)
if(STRATA_Q38_FASTMATH)
    list(APPEND Q38_NVCC_EXTRA --use_fast_math)
endif()
if(STRATA_Q38_REQUIRE_CUDA135)
    if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 13.5 OR NOT CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 13.6)
        message(FATAL_ERROR "This candidate requires the CUDA 13.5 compiler")
    endif()
    if(CUDAToolkit_VERSION VERSION_LESS 13.5 OR NOT CUDAToolkit_VERSION VERSION_LESS 13.6)
        message(FATAL_ERROR "This candidate requires the CUDA 13.5 Toolkit")
    endif()
endif()
get_filename_component(ROOT "../.." ABSOLUTE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${ROOT}/data/spin8/reference.profile)
file(READ ${ROOT}/data/spin8/reference.profile Q38_PROFILE_HEX HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," Q38_PROFILE_BYTES "${Q38_PROFILE_HEX}")
configure_file(reference_profile.hpp.in reference_profile.hpp @ONLY)
add_library(q38_spin8 STATIC ${ROOT}/spin8/src/spin8.cpp ${ROOT}/spin8/third_party/sha256/sha256.c)
target_include_directories(q38_spin8 PUBLIC ${ROOT}/spin8/include ${ROOT}/spin8/third_party/sha256)
target_compile_options(q38_spin8 PRIVATE /fp:strict)
add_library(q38_ops STATIC ops.cu prefill_gemm.cu ${ROOT}/src/kernels/cuda/fused_gdn.cu)
target_include_directories(q38_ops PUBLIC ${ROOT}/include ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(q38_ops PUBLIC NOMINMAX)
target_link_libraries(q38_ops PUBLIC CUDA::cudart CUDA::cublas CUDA::cublasLt)
if(STRATA_Q38_EXPERIMENTAL_FUSED_WMMA)
    target_compile_definitions(q38_ops PRIVATE STRATA_Q38_FUSED_WMMA=1)
endif()
add_library(q38_kv STATIC kv.cu kv_config.cpp)
target_include_directories(q38_kv PRIVATE ${ROOT}/tqcodec/src ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(q38_kv PUBLIC q38_ops q38_spin8)
target_compile_options(q38_kv PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--ftz=false;--prec-div=true;--prec-sqrt=true> ${Q38_NVCC_EXTRA})
target_compile_options(q38_ops PRIVATE ${Q38_NVCC_EXTRA})
add_executable(strata-q38 engine.cu)
target_compile_options(strata-q38 PRIVATE ${Q38_NVCC_EXTRA})
target_link_libraries(strata-q38 PRIVATE q38_kv psapi)
add_executable(q38-numerics numerics.cu)
target_link_libraries(q38-numerics PRIVATE q38_ops psapi)
add_executable(q38-protocol-test protocol_test.cpp)
target_compile_features(q38-protocol-test PRIVATE cxx_std_17)
enable_testing()
add_executable(q38-input-reader-test input_reader_test.cpp)
target_compile_definitions(q38-input-reader-test PRIVATE NOMINMAX)
add_test(NAME input_reader COMMAND q38-input-reader-test)
set_tests_properties(input_reader PROPERTIES TIMEOUT 15)
add_executable(q38-language-test language_test.cpp)
target_include_directories(q38-language-test PRIVATE ${ROOT}/include)
add_test(NAME language_cpu COMMAND q38-language-test)
add_test(NAME protocol COMMAND q38-protocol-test)
add_test(NAME numerics COMMAND q38-numerics)
add_executable(q38-iq-cpu iq_cpu_test.cpp)
target_include_directories(q38-iq-cpu PRIVATE ${ROOT}/include)
add_test(NAME iq_cpu COMMAND q38-iq-cpu)
add_executable(q38-iq-gpu iq_gpu_test.cu)
target_link_libraries(q38-iq-gpu PRIVATE q38_ops psapi)
add_test(NAME iq_gpu COMMAND q38-iq-gpu)
add_executable(q38-reader-regression ${ROOT}/src/artifact/gguf_reader_test.cpp)
target_include_directories(q38-reader-regression PRIVATE ${ROOT}/include)
add_executable(q38-geometry-regression ${ROOT}/tests/core/gguf_geometry_test.cpp)
target_include_directories(q38-geometry-regression PRIVATE ${ROOT}/include)
add_test(NAME reader_regression COMMAND q38-reader-regression)
add_test(NAME geometry_regression COMMAND q38-geometry-regression)
add_executable(q38-family-regression ${ROOT}/tests/core/model_family_test.cpp)
target_include_directories(q38-family-regression PRIVATE ${ROOT}/include)
add_test(NAME family_regression COMMAND q38-family-regression)
add_executable(q38-reader-metadata reader_metadata_test.cpp)
target_include_directories(q38-reader-metadata PRIVATE ${ROOT}/include)
add_test(NAME reader_metadata COMMAND q38-reader-metadata)
add_executable(q38-geometry geometry_test.cpp)
target_include_directories(q38-geometry PRIVATE ${ROOT}/include)
add_test(NAME q38_geometry COMMAND q38-geometry)
set_tests_properties(q38_geometry PROPERTIES TIMEOUT 60)
add_library(q38_dialect_reference STATIC ${ROOT}/tqcodec/src/dialects.cpp)
target_include_directories(q38_dialect_reference PUBLIC ${ROOT}/tqcodec/include)
target_compile_options(q38_dialect_reference PRIVATE /fp:strict)
add_executable(q38-kv-numerics kv_numerics.cu)
target_include_directories(q38-kv-numerics PRIVATE ${ROOT}/tqcodec/src)
target_link_libraries(q38-kv-numerics PRIVATE q38_kv q38_dialect_reference psapi)
target_compile_options(q38-kv-numerics PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--ftz=false;--prec-div=true;--prec-sqrt=true>)
add_test(NAME kv_numerics COMMAND q38-kv-numerics)
add_executable(q38-kv-cpu kv_cpu_test.cpp)
target_link_libraries(q38-kv-cpu PRIVATE q38_kv)
add_test(NAME kv_cpu COMMAND q38-kv-cpu ${ROOT}/data/spin8/reference.profile)
add_executable(q38-hardware-test hardware_test.cu)
target_link_libraries(q38-hardware-test PRIVATE q38_kv psapi)
add_test(NAME hardware COMMAND q38-hardware-test)
add_executable(q38-prefill-gemm-test prefill_gemm_test.cu)
target_link_libraries(q38-prefill-gemm-test PRIVATE q38_ops)
add_test(NAME prefill_gemm COMMAND q38-prefill-gemm-test)

target_link_libraries(q38-fused-wmma-test PRIVATE q38_ops)
add_test(NAME fused_wmma COMMAND q38-fused-wmma-test)

add_test(NAME spin8_triality COMMAND q38-spin8-triality-test)

target_include_directories(q38_dialect_cuda PUBLIC ${ROOT}/tqcodec/include)
target_link_libraries(q38_dialect_cuda PUBLIC CUDA::cudart)
target_compile_options(q38_dialect_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--ftz=false;--prec-div=true;--prec-sqrt=true>)
add_executable(q38-dialect-cuda-regression ${ROOT}/tqcodec/tests/dialects_cuda_test.cpp)
target_link_libraries(q38-dialect-cuda-regression PRIVATE q38_dialect_cuda q38_dialect_reference)
