```python
"""
jobqueue.py - Persistent, priority-aware job queue for a single machine.

Provides a SQLite-backed job queue with priority scheduling, fair-share
distribution among submitters, retry with exponential backoff, and a
multi-threaded worker pool with lease-based recovery.
"""

from __future__ import annotations

import argparse
import json
import logging
import sqlite3
import threading
import time
import uuid
from dataclasses import dataclass, field, asdict
from datetime import datetime, timezone
from enum import Enum
from typing import Any, Callable, Optional

logger = logging.getLogger(__name__)


class JobState(str, Enum):
    QUEUED = "queued"
    RUNNING = "running"
    DONE = "done"
    FAILED = "failed"
    CANCELLED = "cancelled"


@dataclass
class Job:
    """Represents a single job in the queue."""

    id: str = field(default_factory=lambda: str(uuid.uuid4()))
    kind: str = ""
    priority: int = 100
    payload: dict = field(default_factory=dict)
    state: str = JobState.QUEUED.value
    attempts: int = 0
    max_attempts: int = 3
    created_at: str = field(default_factory=lambda: datetime.now(timezone.utc).isoformat())
    started_at: Optional[str] = None
    finished_at: Optional[str] = None
    error: Optional[str] = None

    def to_dict(self) -> dict:
        return asdict(self)

    @classmethod
    def from_row(cls, row: tuple) -> "Job":
        """Construct a Job from a SQLite row tuple."""
        return cls(
            id=row[0],
            kind=row[1],
            priority=row[2],
            payload=json.loads(row[3]) if row[3] else {},
            state=row[4],
            attempts=row[5],
            max_attempts=row[6],
            created_at=row[7],
            started_at=row[8],
            finished_at=row[9],
            error=row[10],
        )


class JobQueue:
    """
    Persistent, thread-safe, priority-aware job queue backed by SQLite.

    Uses WAL mode for concurrent read/write performance. Supports atomic
    claiming of jobs with fair-share round-robin among submitters at equal
    priority.
    """

    def __init__(self, path: str) -> None:
        """
        Initialize the job queue with a SQLite database at the given path.

        Args:
            path: Filesystem path to the SQLite database file.
        """
        self._path = path
        self._lock = threading.Lock()
        self._conn = sqlite3.connect(path, check_same_thread=False)
        self._conn.execute("PRAGMA journal_mode=WAL")
        self._conn.execute("PRAGMA busy_timeout=5000")
        self._create_tables()

    def _create_tables(self) -> None:
        """Create the jobs table if it does not exist."""
        with self._lock:
            self._conn.execute("""
                CREATE TABLE IF NOT EXISTS jobs (
                    id TEXT PRIMARY KEY,
                    kind TEXT NOT NULL,
                    priority INTEGER NOT NULL,
                    payload TEXT NOT NULL,
                    state TEXT NOT NULL,
                    attempts INTEGER NOT NULL DEFAULT 0,
                    max_attempts INTEGER NOT NULL DEFAULT 3,
                    created_at TEXT NOT NULL,
                    started_at TEXT,
                    finished_at TEXT,
                    error TEXT,
                    lease_expires_at TEXT,
                    submitter TEXT
                )
            """)
            self._conn.execute("""
                CREATE INDEX IF NOT EXISTS idx_jobs_state_priority
                ON jobs (state, priority, created_at)
            """)
            self._conn.execute("""
                CREATE INDEX IF NOT EXISTS idx_jobs_lease
                ON jobs (state, lease_expires_at)
            """)
            self._conn.commit()

    def _get_conn(self) -> sqlite3.Connection:
        """Return the shared connection (thread-safe via lock usage)."""
        return self._conn

    def submit(
        self,
        kind: str,
        payload: dict,
        priority: int = 100,
        max_attempts: int = 3,
    ) -> Job:
        """
        Submit a new job to the queue.

        Args:
            kind: The type/category of the job.
            payload: Arbitrary JSON-serializable data for the job.
            priority: Lower values run first. Default 100.
            max_attempts: Maximum number of execution attempts. Default 3
