def silence_points(audio: np.ndarray, sr: int, min_silence_s: float = 0.25,
                   merge_gap_s: float = 0.1) -> list[float]:
    """Midpoints of the quiet stretches in `audio`, in seconds.

    Silences shorter than `min_silence_s` are ignored. Silences separated by a gap of
    less than `merge_gap_s` are merged into one before their midpoints are returned.
    """
    audio = np.asarray(audio, dtype=np.float32).reshape(-1)
    hop = max(1, sr // 100)
    frames = len(audio) // hop
    if frames < 3:
        return []
    rms = np.sqrt(np.maximum(1e-12, np.mean(
        audio[:frames * hop].reshape(frames, hop) ** 2, axis=1)))
    db = 20.0 * np.log10(rms / max(float(np.max(rms)), 1e-9) + 1e-12)
    quiet = db < SILENCE_DB

    # Find raw silence intervals (start_frame, end_frame)
    intervals: list[tuple[int, int]] = []
    i = 0
    while i < frames:
        if not quiet[i]:
            i += 1
            continue
        j = i
        while j < frames and quiet[j]:
            j += 1
        intervals.append((i, j))
        i = j

    # Filter by minimum duration
    min_frames = int(round(min_silence_s * sr / hop))
    intervals = [(s, e) for s, e in intervals if (e - s) * hop / sr >= min_silence_s]

    if not intervals:
        return []

    # Merge intervals separated by less than merge_gap_s
    merged: list[tuple[int, int]] = []
    for s, e in intervals:
        if merged:
            prev_s, prev_e = merged[-1]
            gap_s = (s - prev_e) * hop / sr
            if gap_s < merge_gap_s:
                merged[-1] = (prev_s, e)
                continue
        merged.append((s, e))
