def silence_points(audio: np.ndarray, sr: int, min_silence_s: float = 0.25,
                   merge_gap_s: float = 0.1) -> list[float]:
    """Midpoints of the quiet stretches in `audio`, in seconds.

    Silences shorter than `min_silence_s` are ignored. Silences separated by a
    non-silent gap shorter than `merge_gap_s` are merged into one before their
    midpoints are returned.
    """
    audio = np.asarray(audio, dtype=np.float32).reshape(-1)
    hop = max(1, sr // 100)
    frames = len(audio) // hop
    if frames < 3:
        return []
    rms = np.sqrt(np.maximum(1e-12, np.mean(
        audio[:frames * hop].reshape(frames, hop) ** 2, axis=1)))
    db = 20.0 * np.log10(rms / max(float(np.max(rms)), 1e-9) + 1e-12)
    quiet = db < SILENCE_DB

    # Collect raw silence runs as (start_frame, end_frame)
    runs: list[tuple[int, int]] = []
    i = 0
    while i < frames:
        if not quiet[i]:
            i += 1
            continue
        j = i
        while j < frames and quiet[j]:
            j += 1
        runs.append((i, j))
        i = j

    # Merge runs separated by a gap shorter than merge_gap_s
    merged: list[tuple[int, int]] = []
    for start, end in runs:
        if merged and (start - merged[-1][1]) * hop / sr < merge_gap_s:
            merged[-1] = (merged[-1][0], end)
        else:
            merged.append((start, end))

    # Filter by min_silence_s and compute midpoints
    out: list[float] = []
    for start, end in merged:
        if (end - start) * hop / sr >= min_silence_s:
            out.append((start + end) * 0.5 * hop / sr)
    return out
