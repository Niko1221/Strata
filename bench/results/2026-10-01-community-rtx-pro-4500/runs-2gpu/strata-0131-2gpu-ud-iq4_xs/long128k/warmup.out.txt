                   merge_gap_s: float = 0.1) -> list[float]:
    """Midpoints of the quiet stretches in `audio`, in seconds.

    Silences shorter than `min_silence_s` are ignored. Silences separated by a gap
    shorter than `merge_gap_s` are merged into one before their midpoints are returned.
    """

    hop = max(1, sr // 100)
    frames = len(audio) // hop
    if frames < 3:
        return []

        audio[:frames * hop].reshape(frames, hop) ** 2, axis=1)))

    raw_intervals: list[tuple[int, int]] = []

        if not quiet[i]:
            i += 1
            continue

            j += 1
        raw_intervals.append((i, j))
        i = j

    if not raw_intervals:
        return []
