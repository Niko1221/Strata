"""
jobqueue.py - A persistent, priority-aware job queue for a single machine.

This module provides a thread-safe job queue backed by SQLite (WAL mode),
with support for priority scheduling, fair-share round-robin among submitters,
exponential backoff retries, and a multi-threaded worker pool with lease-based
recovery for dead workers.
"""

from __future__ import annotations

import argparse
import json
import sqlite3
import threading
import time
import traceback
import uuid
from dataclasses import dataclass, field
from datetime import datetime, timezone
from enum import Enum
from typing import Any, Callable, Optional


class JobState(str, Enum):
    """Enumeration of possible job states."""
    QUEUED = "queued"
    RUNNING = "running"
    DONE = "done"
    FAILED = "failed"
    CANCELLED = "cancelled"


@dataclass
class Job:
    """
    Represents a single job in the queue.

    Attributes:
        id: Unique identifier (uuid4 string).
        kind: Type/category of the job.
        priority: Integer priority; lower values run first.
        payload: Arbitrary dictionary of job data.
        state: Current state of the job.
        attempts: Number of times this job has been attempted.
        max_attempts: Maximum number of attempts before marking as failed.
        created_at: ISO format timestamp when the job was created.
        started_at: ISO format timestamp when the job started running, or None.
        finished_at: ISO format timestamp when the job finished, or None.
        error: Error message if the job failed, or None.
    """
    id: str
    kind: str
    priority: int
    payload: dict
    state: str
    attempts: int
    max_attempts: int
    created_at: str
    started_at: Optional[str] = None
    finished_at: Optional[str] = None
    error: Optional[str] = None

    def to_row(self) -> tuple:
        """Convert job to a tuple suitable for SQLite insertion."""
        return (
            self.id,
            self.kind,
            self.priority,
            json.dumps(self.payload),
            self.state,
            self.attempts,
            self.max_attempts,
            self.created_at,
            self.started_at,
            self.finished_at,
            self.error,
        )

    @classmethod
    def from_row(cls, row: tuple) -> "Job":
        """Create a Job instance from a SQLite row tuple."""
        return cls(
            id=row[0],
            kind=row[1],
            priority=row[2],
            payload=json.loads(row[3]),
            state=row[4],
            attempts=row[5],
            max_attempts=row[6],
            created_at=row[7],
            started_at=row[8],
            finished_at=row[9],
            error=row[10],
        )


def _now_iso() -> str:
    """Return the current UTC time as an ISO format string."""
    return datetime.now(timezone.utc).isoformat()


def _parse_iso(ts: str) -> datetime:
    """Parse an ISO format timestamp string to a datetime object."""
    return datetime.fromisoformat(ts)


class JobQueue:
    """
    A persistent, thread-safe, priority-aware job queue backed by SQLite.

    Jobs are stored in a SQLite database with WAL mode enabled for better
    concurrent read performance. The queue supports priority scheduling,
    fair-share round-robin among submitters (identified by payload["by"]),
    and exponential backoff retries.
    """

    def __init__(self, path: str) -> None:
        """
        Initialize the job queue.

        Args:
            path: Filesystem path to the SQLite database file.
        """
        self._path = path
        self._lock = threading.Lock()
        self._conn = sqlite3.connect(path, check_same_thread=False)
        self._conn.execute("PRAGMA journal_mode=WAL")
        self._conn.execute("PRAGMA busy_timeout=5000")
        self._create_tables()

    def _create_tables(self) -> None:
        """Create the jobs table if it does not exist."""
        with self._lock:

                CREATE TABLE IF NOT EXISTS jobs (
                    id TEXT PRIMARY KEY,
                    kind TEXT NOT NULL,
                    priority INTEGER NOT NULL,
                    payload TEXT NOT NULL,
                    state TEXT NOT NULL,
                    attempts INTEGER NOT NULL DEFAULT 0,
                    max_attempts INTEGER NOT NULL DEFAULT 3,
                    created_at TEXT NOT NULL
