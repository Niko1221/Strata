class LRUCache:
    """
    A bounded, thread-safe Least Recently Used (LRU) cache.
    
    This cache evicts the least recently used items when the capacity is exceeded.
    All operations are thread-safe using a reentrant lock.
    
    Attributes:
        capacity (int): Maximum number of items the cache can hold.
        _cache (dict): Internal storage mapping keys to values.
        _order (collections.OrderedDict): Maintains insertion/access order for LRU eviction.
        _lock (threading.RLock): Reentrant lock for thread safety.
        _hits (int): Counter for cache hits.
        _misses (int): Counter for cache misses.
        _evictions (int): Counter for evicted items.
    """
    
    def __init__(self, capacity: int):
        """
        Initialize the LRU cache with a given capacity.
        
        Args:
            capacity (int): Maximum number of items the cache can hold. Must be positive.
            
        Raises:
            ValueError: If capacity is not a positive integer.
            TypeError: If capacity is not an integer.
        """
        if not isinstance(capacity, int):
            raise TypeError(f"capacity must be an integer, got {type(capacity).__name__}")
        if capacity <= 0:
            raise ValueError(f"capacity must be a positive integer, got {capacity}")
        
        self.capacity = capacity
        self._cache: Dict[Any, Any] = {}
        self._order: collections.OrderedDict = collections.OrderedDict()
        self._lock = threading.RLock()
        self._hits = 0
        self._misses = 0
        self._evictions = 0
    
    def get(self, key: Any, default: Any = None) -> Any:
        """
        Retrieve a value from the cache.
        
        If the key exists, it is moved to the end (most recently used) and returned.
        If the key does not exist, the default value is returned and a miss is recorded.
        
        Args:
            key (Any): The key to look up.
            default (Any): Value to return if key is not found. Defaults to None.
            
        Returns:
            Any: The cached value if found, otherwise the default value.
        """
        with self._lock:
            if key in self._cache:
                self._hits += 1
                # Move to end (most recently used)
                self._order.move_to_end(key)
                return self._cache[key]
            else:
                self._misses += 1
                return default
    
    def put(self, key: Any, value: Any) -> None:
        """
        Insert or update a key-value pair in the cache.
        
        If the key already exists, its value is updated and it is moved to the end.
        If the cache is at capacity, the least recently used item is evicted.
        
        Args:
            key (Any): The key to insert or update.
            value (Any): The value to associate with the key.
        """
        with self._lock:
            if key in self._cache:
                # Update existing key and move to end
                self._cache[key] = value
                self._order.move_to_end(key)
            else:

                if len(self._cache) >= self.capacity:
