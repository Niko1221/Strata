class LRUCache:
    """
    A bounded Least Recently Used (LRU) cache implementation.
    
    This cache stores key-value pairs and evicts the least recently used items
    when the cache exceeds its maximum capacity. It uses a doubly linked list
    (via collections.OrderedDict) to maintain access order and a dictionary
    for O(1) lookups.
    
    Attributes:
        capacity (int): The maximum number of items the cache can hold.
        _cache (collections.OrderedDict): Internal storage for cache entries.
        _lock (threading.Lock): Lock for thread-safe operations.
    """
    
    def __init__(self, capacity: int):
        """
        Initialize the LRU Cache with a given capacity.
        
        Args:
            capacity (int): Maximum number of items to store. Must be positive.
            
        Raises:
            ValueError: If capacity is not a positive integer.
        """
        if not isinstance(capacity, int) or capacity <= 0:
            raise ValueError("Capacity must be a positive integer")
        self.capacity = capacity
        self._cache = collections.OrderedDict()
        self._lock = threading.Lock()
        self._hits = 0
        self._misses = 0

    def get(self, key: Any) -> Optional[Any]:
        """
        Retrieve a value from the cache.
        
        If the key exists, it is moved to the end (most recently used) and returned.
        If the key does not exist, None is returned.
        
        Args:
            key (Any): The key to look up.
            
        Returns:
            Optional[Any]: The value associated with the key, or None if not found.
        """
        with self._lock:
            if key in self._cache:
                self._cache.move_to_end(key)
                self._hits += 1
                return self._cache[key]
            self._misses += 1
            return None

    def put(self, key: Any, value: Any) -> None:
        """
        Insert or update a key-value pair in the cache.
        
        If the key already exists, its value is updated and it is moved to the end.
        If the key does not exist, it is added. If the cache exceeds capacity,
        the least recently used item is evicted.
        
        Args:
            key (Any): The key to insert or update.
            value (Any): The value to associate with the key.
        """
        with self._lock:
            if key in self._cache:
                self._cache.move_to_end(key)
                self._cache[key] = value
            else:
                if len(self._cache) >= self.capacity:
                    # Evict least recently used (first item in OrderedDict)
                    self._cache.popitem(last=False)
                self._cache[key] = value
                self._cache.move_to_end(key)

    def delete(self, key: Any) -> bool:
        """
        Remove a key from the cache.
        
        Args:
            key (Any): The key to remove.
            
        Returns:
            bool: True if the key was removed, False if it did not exist.
        """
        with self._lock:
            if key in self._cache:
                del self._cache[key]
                return True
            return False

    def clear(self) -> None:
        """
        Remove all items from the cache.
        """
        with self._lock:
            self._cache.clear()
            self._hits = 0
            self._misses = 0

    def size(self) -> int:
        """
        Get the current number of items in the cache.
        
        Returns:
            int: Number of items currently stored.
        """
        with self._lock:
            return len(self._cache)

    def stats(self) -> Dict[str, int]:
        """
        Get cache statistics.
        
        Returns:
            Dict[str, int]: Dictionary containing 'hits', 'misses', and 'size'.
        """
        with self._lock:
            return {
                'hits': self._hits,
                'misses': self._misses,
                'size': len(self._cache)
            }

    def __contains__(self, key: Any) -> bool:
        """
        Check if a key exists in the cache without updating access order.
        
        Args:
            key (Any): The key to check.
            
        Returns:
            bool: True if key exists, False otherwise.
        """
        with self._lock:
            return key in self._cache

    def __len__(self) -> int:
        """
        Return the number of items in the cache.
        
        Returns:
            int: Number of items.
        """
        return self.size()

    def __repr__(self) -> str:
        """
        String representation of the cache.
        
        Returns:
            str: Representation showing capacity and current size.
        """
        return f"LRUCache(capacity={self.capacity}, size={self.size()})"

        self.cache.get('a')

        self.cache.put('d', 4)
        self.assertIsNone(self.cache.get('b'))
        self.assertEqual(self.cache.get('a'), 1)
        self.assertEqual(self.cache.get('c'), 3)

        self.cache.get('b')  # b is now MRU, a is LRU among existing
        self.cache.put('d', 4)  # Should evict 'c' (LRU)
        self.assertIsNone(self.cache.get('c'))
        self.assertEqual(self.cache.get('a'), 1)
        self.assertEqual(self.cache.get('b'), 2)
        self.assertEqual(self.cache.get('d'), 4)
