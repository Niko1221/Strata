class LRUCache:
    """
    A bounded Least Recently Used (LRU) cache implementation.
    
    This cache stores key-value pairs and evicts the least recently used items
    when the cache exceeds its maximum capacity. It uses an OrderedDict to
    maintain the order of access.
    
    Attributes:
        capacity (int): The maximum number of items the cache can hold.
        _cache (OrderedDict): Internal storage for cache items.
        _lock (threading.Lock): Lock for thread-safe operations.
    """

    def __init__(self, capacity: int):
        """
        Initialize the LRU Cache.
        
        Args:
            capacity (int): The maximum number of items the cache can hold.
                            Must be a positive integer.
        
        Raises:
            ValueError: If capacity is not a positive integer.
        """
        if not isinstance(capacity, int) or capacity <= 0:
            raise ValueError("Capacity must be a positive integer.")
        self.capacity = capacity
        self._cache = OrderedDict()
        self._lock = threading.Lock()

    def get(self, key: Any) -> Optional[Any]:
        """
        Retrieve an item from the cache.
        
        If the key exists, it is moved to the end of the OrderedDict to mark
        it as recently used.
        
        Args:
            key (Any): The key to look up.
        
        Returns:
            Optional[Any]: The value associated with the key, or None if not found.
        """
        with self._lock:
            if key in self._cache:
                # Move to end to mark as recently used
                self._cache.move_to_end(key)
                return self._cache[key]
            return None

    def put(self, key: Any, value: Any) -> None:
        """
        Insert or update an item in the cache.
        
        If the key already exists, its value is updated and it is moved to the
        end. If the cache is full, the least recently used item is evicted.
        
        Args:
            key (Any): The key to insert or update.
            value (Any): The value to associate with the key.
        """
        with self._lock:
            if key in self._cache:
                # Update value and move to end
                self._cache[key] = value
                self._cache.move_to_end(key)
            else:
                # Check capacity
                if len(self._cache) >= self.capacity:
                    # Evict least recently used (first item)
                    self._cache.popitem(last=False)
                # Insert new item
                self._cache[key] = value

    def delete(self, key: Any) -> bool:
        """
        Remove an item from the cache.
        
        Args:
            key (Any): The key to remove.
        
        Returns:
            bool: True if the key was found and removed, False otherwise.
        """
        with self._lock:
            if key in self._cache:
                del self._cache[key]
                return True
            return False

    def clear(self) -> None:
        """
        Remove all items from the cache.
        """
        with self._lock:
            self._cache.clear()

    def size(self) -> int:
        """
        Get the current number of items in the cache.
        
        Returns:
            int: The number of items currently in the cache.
        """
        with self._lock:
            return len(self._cache)

    def contains(self, key: Any) -> bool:
        """
        Check if a key exists in the cache without updating its usage status.
        
        Args:
            key (Any): The key to check.
        
        Returns:
            bool: True if the key exists, False otherwise.
        """
        with self._lock:
            return key in self._cache

    def keys(self) -> List[Any]:
        """
        Get a list of all keys in the cache, ordered from least to most recently used.
        
        Returns:
            List[Any]: A list of keys.
        """
        with self._lock:
            return list(self._cache.keys())

    def values(self) -> List[Any]:
        """
        Get a list of all values in the cache, ordered from least to most recently used.
        
        Returns:
            List[Any]: A list of values.
        """
        with self._lock:
            return list(self._cache.values())

    def items(self) -> List[Tuple[Any, Any]]:
        """
        Get a list of all key-value pairs in the cache, ordered from least to most recently used.
        
        Returns:
            List[Tuple[Any, Any]]: A list of (key, value) tuples.
        """
        with self._lock:
            return list(self._cache.items())

    def __len__(self) -> int:
        """
        Return the number of items in the cache.
        
        Returns:
            int: The number of items.
        """
        return self.size()

    def __contains__(self, key: Any) -> bool:
        """
        Check if a key exists in the cache.
        
        Args:
            key (Any): The key to check.
        
        Returns:
            bool: True if the key exists, False otherwise.
        """
        return self.contains(key)

    def __getitem__(self, key: Any) -> Any:
        """
        Retrieve an item from the cache using bracket notation.
        
        Args:
            key (Any): The key to look up.
        
        Returns:
            Any: The value associated with the key.
        
        Raises:
            KeyError: If the key is not found.
        """
        value = self.get(key)
        if value is None and key not in self._cache:
            raise KeyError(f"Key {key} not found in cache")
        return value

    def __setitem__(self, key: Any, value: Any) -> None:
        """
        Insert or update an item in the cache using bracket notation.
        
        Args:
            key (Any): The key to insert or update.
            value (Any): The value to associate with the key.
        """
        self.put(key, value)

    def __delitem__(self, key: Any) -> None:
        """
        Remove an item from the cache using bracket notation.
        
        Args:
            key (Any): The key to remove.
        
        Raises:
            KeyError: If the key is not found.
        """
        if not self.delete(key):
            raise KeyError(f"Key {key} not found in cache")

def retry(max_attempts: int = 3, delay: float = 1.0, backoff_factor: float = 2.0, 
          exceptions: Tuple[type, ...] = (Exception,), 
          on_failure: Optional[Callable[[Exception, int], None]] = None):
    """
    A decorator that retries a function call if it raises specified exceptions.
    
    Args:
        max_attempts (int): Maximum number of attempts.
        delay (float): Initial delay between retries in seconds.
        backoff_factor (float): Multiplier for delay after each retry.
        exceptions (Tuple[type, ...]): Tuple of exception types to catch and retry.
        on_failure (Optional[Callable[[Exception, int], None]]): Callback called on each failure.
    
    Returns:
        Callable: The decorated function.
    """
    def decorator(func: Callable) -> Callable:
        @wraps(func)
        def wrapper(*args, **kwargs):
            current_delay = delay
            last_exception = None
            
            for attempt in range(1, max_attempts + 1):
                try:
                    return func(*args, **kwargs)
                except exceptions as e:
                    last_exception = e
                    if on_failure:
                        on_failure(e, attempt)
                    
                    if attempt < max_attempts:
                        time.sleep(current_delay)
                        current_delay *= backoff_factor
                    else:
                        raise last_exception
            
            # Should not reach here if max_attempts >= 1
            raise last_exception
        
        return wrapper
    return decorator


class TestRetryDecorator(unittest.TestCase):
    """Tests for the retry decorator."""

    def setUp(self):
        """Set up test fixtures."""
        self.call_count = 0
        self.failures = []

    def _reset_counters(self):
        """Reset counters for each test."""
        self.call_count = 0
        self.failures = []

    def test_success_first_attempt(self):
        """Test function succeeds on first attempt."""
        self._reset_counters()
        
        @retry(max_attempts=3, delay=0.01)

            self.call_count += 1
            return "success"
        
        result = success_func()

        self.assertEqual(self.call_count, 1)

    def test_failure_then_success(self):

            always_fail()

        self.assertEqual(len(self.failures), 3)

        @retry(max_attempts=2, delay=0.01)
        def inner():
            self.call_count += 1
            if self.call_count < 3:
                raise ValueError("Inner fail")
            return "inner_success"
        
        @retry(max_attempts=2, delay=0.01)
        def outer():
            return inner()
        
        result = outer()
        self.assertEqual(result, "inner_success")
        # Inner called twice by outer's first attempt (fails), then once more by outer's second attempt
        # Actually: outer attempt 1 -> inner attempt 1 (fail), inner attempt 2 (fail) -> outer fails
        # outer attempt 2 -> inner attempt 1 (success)
        # Total inner calls: 3
        self.assertEqual(self.call_count, 3)

    def test_on_failure_none(self):
        """Test on_failure=None works correctly."""
        self._reset_counters()
        
        @retry(max_attempts=3, delay=0.01, on_failure=None)
        def always_fail():
            self.call_count += 1
            raise ValueError("Fail")
        
        with self.assertRaises(ValueError):
            always_fail()
        self.assertEqual(self.call_count, 3)

    def test_default_parameters(self):

    def test_exception_hierarchy(self):
        """Test that catching base exception catches subclasses."""
        self._reset_counters()
        
        class CustomError(Exception):
            pass
        
        @retry(max_attempts=3, delay=0.01, exceptions=(Exception,))
        def raise_custom():
            self.call_count += 1
            if self.call_count < 2:
                raise CustomError("Custom")
            return "success"
        
        result = raise_custom()
        self.assertEqual(result, "success")
        self.assertEqual(self.call_count, 2)

    def test_return_value_type(self):

        self.assertIsInstance(result, dict)
        self.assertEqual(result, {"key": "value"})

    def test_generator_function_not_retried(self):
        """Test that generator functions are not retried (they return immediately)."""
        self._reset_counters()
        
        @retry(max_attempts=3, delay=0.01)
        def gen_func():
            self.call_count += 1
            yield 1
            yield 2
        
        gen = gen_func()
        # The function call itself succeeds, returning a generator
        self.assertEqual(self.call_count, 1)
        # Consuming the generator doesn't trigger retry
        items = list(gen)
        self.assertEqual(items, [1, 2])
        self.assertEqual(self.call_count, 1)

    def test_max_attempts_zero(self):
        """Test max_attempts=0 raises ValueError or behaves unexpectedly."""
        # With max_attempts=0, the loop doesn't execute, and last_exception is None
        # This might raise UnboundLocalError or similar. Let's test behavior.
        self._reset_counters()
        
        @retry(max_attempts=0, delay=0.01)
        def always_fail():
            self.call_count += 1
            raise ValueError("Fail")
        
        # Depending on implementation, this might raise the exception or an error
        # Our implementation: loop range(1, 1) is empty, last_exception is None, raise None -> TypeError
        with self.assertRaises(TypeError):

    def test_negative_max_attempts(self):

    def test_on_failure_raises_exception(self):
        """Test if on_failure callback raises an exception."""
        self._reset_counters()
        
        def bad_on_failure(exc, attempt):
            raise RuntimeError("Callback error")
        
        @retry(max_attempts=3, delay=0.01, on_failure=bad_on_failure)
        def always_fail():
            self.call_count += 1
            raise ValueError("Fail")
        
        with self.assertRaises(RuntimeError):
            always_fail()
        self.assertEqual(self.call_count, 1)  # First failure triggers callback which raises

    def test_multiple_decorators(self):
        """Test retry with other decorators."""
        self._reset_counters()
        
        def log_decorator(func):
            @wraps(func)
            def wrapper(*args, **kwargs):
                self.call_count += 1
                return func(*args, **kwargs)
            return wrapper
        
        @retry(max_attempts=3, delay=0.01)
        @log_decorator
        def always_fail():
            raise ValueError("Fail")
        
        with self.assertRaises(ValueError):
            always_fail()
        # log_decorator increments call_count on each call to the wrapped function
        # retry calls the wrapped function 3 times
        self.assertEqual(self.call_count, 3)
