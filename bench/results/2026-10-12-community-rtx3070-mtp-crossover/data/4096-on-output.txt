class LRUCache:
    """
    A thread-safe, bounded Least Recently Used (LRU) cache.
    
    This cache evicts the least recently used items when the capacity is exceeded.
    It uses a doubly-linked list (via OrderedDict) for O(1) access and eviction.
    
    Attributes:
        capacity (int): The maximum number of items the cache can hold.
        _cache (collections.OrderedDict): The underlying storage for cache items.
        _lock (threading.Lock): A lock to ensure thread safety.
    """
    
    def __init__(self, capacity: int):
        """
        Initialize the LRU Cache.
        
        Args:
            capacity (int): The maximum number of items to store. Must be positive.
            
        Raises:
            ValueError: If capacity is not a positive integer.
        """
        if not isinstance(capacity, int) or capacity <= 0:
            raise ValueError("Capacity must be a positive integer.")
        self.capacity = capacity
        self._cache = collections.OrderedDict()
        self._lock = threading.Lock()
        self._hits = 0
        self._misses = 0

    def get(self, key: Any) -> Optional[Any]:
        """
        Retrieve an item from the cache.
        
        If the key exists, it is moved to the end (most recently used).
        
        Args:
            key (Any): The key to look up.
            
        Returns:
            Optional[Any]: The value associated with the key, or None if not found.
        """
        with self._lock:
            if key in self._cache:
                self._cache.move_to_end(key)
                self._hits += 1
                return self._cache[key]
            self._misses += 1
            return None

    def put(self, key: Any, value: Any) -> None:
        """
        Insert or update an item in the cache.
        
        If the key already exists, its value is updated and it is moved to the end.
        If the cache is full, the least recently used item is evicted.
        
        Args:
            key (Any): The key to insert or update.
            value (Any): The value to associate with the key.
        """
        with self._lock:
            if key in self._cache:
                self._cache.move_to_end(key)
                self._cache[key] = value
            else:
                if len(self._cache) >= self.capacity:
                    # Evict least recently used (first item in OrderedDict)
                    self._cache.popitem(last=False)
                self._cache[key] = value

    def delete(self, key: Any) -> bool:
        """
        Remove an item from the cache.
        
        Args:
            key (Any): The key to remove.
            
        Returns:
            bool: True if the key was found and removed, False otherwise.
        """
        with self._lock:
            if key in self._cache:
                del self._cache[key]
                return True
            return False

    def clear(self) -> None:
        """Clear all items from the cache."""
        with self._lock:
            self._cache.clear()
            self._hits = 0
            self._misses = 0

    def size(self) -> int:
        """
        Get the current number of items in the cache.
        
        Returns:
            int: The number of items currently stored.
        """
        with self._lock:
            return len(self._cache)

    def contains(self, key: Any) -> bool:
        """
        Check if a key exists in the cache without updating its usage status.
        
        Args:
            key (Any): The key to check.
            
        Returns:
            bool: True if the key exists, False otherwise.
        """
        with self._lock:
            return key in self._cache

    def keys(self) -> List[Any]:
        """
        Get a list of all keys in the cache, ordered from least to most recently used.
        
        Returns:
            List[Any]: A list of keys.
        """
        with self._lock:
            return list(self._cache.keys())

    def values(self) -> List[Any]:
        """
        Get a list of all values in the cache, ordered from least to most recently used.
        
        Returns:
            List[Any]: A list of values.
        """
        with self._lock:
            return list(self._cache.values())

    def items(self) -> List[Tuple[Any, Any]]:
        """
        Get a list of all (key, value) pairs in the cache.
        
        Returns:
            List[Tuple[Any, Any]]: A list of tuples.
        """
        with self._lock:
            return list(self._cache.items())

    def get_stats(self) -> Dict[str, int]:
        """
        Get cache statistics.
        
        Returns:
            Dict[str, int]: A dictionary containing 'hits', 'misses', and 'size'.
        """
        with self._lock:
            return {
                "hits": self._hits,
                "misses": self._misses,
                "size": len(self._cache)
            }

    def __len__(self) -> int:
        """Return the number of items in the cache."""
        return self.size()

    def __contains__(self, key: Any) -> bool:
        """Check if a key is in the cache."""
        return self.contains(key)

    def __getitem__(self, key: Any) -> Any:
        """Retrieve an item, raising KeyError if not found."""
        val = self.get(key)
        if val is None and key not in self._cache:
            raise KeyError(key)
        return val

    def __setitem__(self, key: Any, value: Any) -> None:
        """Set an item in the cache."""
        self.put(key, value)

    def __delitem__(self, key: Any) -> None:
        """Delete an item from the cache, raising KeyError if not found."""
        if not self.delete(key):
            raise KeyError(key)

    def test_eviction_lru(self):
        """Test that the least recently used item is evicted."""
        self.cache.put("a", 1)
        self.cache.put("b", 2)
        self.cache.put("c", 3)
        # Access 'a' to make it recently used
        self.cache.get("a")
        # Add 'd', should evict 'b' (LRU)
        self.cache.put("d", 4)
        
        self.assertIsNone(self.cache.get("b"))
        self.assertEqual(self.cache.get("a"), 1)
        self.assertEqual(self.cache.get("c"), 3)
        self.assertEqual(self.cache.get("d"), 4)

    def test_eviction_order(self):
        """Test eviction order with multiple accesses."""
        self.cache.put("a", 1)
        self.cache.put("b", 2)
        self.cache.put("c", 3)
        self.cache.get("a") # a is now MRU, b is LRU
        self.cache.put("d", 4) # evicts b
        
        self.cache.get("c") # c is now MRU, a is LRU
        self.cache.put("e", 5) # evicts a
        
        self.assertIsNone(self.cache.get("a"))
        self.assertIsNone(self.cache.get("b"))
        self.assertEqual(self.cache.get("c"), 3)
        self.assertEqual(self.cache.get("d"), 4)
        self.assertEqual(self.cache.get("e"), 5)

    def test_concurrent_gets_same_key(self):
        """Test concurrent gets on the same key."""
        cache = LRUCache(10)
        cache.put("key", "value")
        results = []
        def reader():
            for _ in range(100):
                results.append(cache.get("key"))
        
        threads = [threading.Thread(target=reader) for _ in range(10)]
        for t in threads:
            t.start()
        for t in threads:
            t.join()
        
        self.assertEqual(len(results), 1000)
        self.assertTrue(all(r == "value" for r in results))

    def test_concurrent_puts_different_keys(self):
        """Test concurrent puts on different keys."""
        cache = LRUCache(100)
        def writer(i):
            cache.put(f"key_{i}", i)
        
        threads = [threading.Thread(target=writer, args=(i,)) for i in range(100)]
        for t in threads:
            t.start()
        for t in threads:
            t.join()
        
        self.assertEqual(cache.size(), 100)
        for i in range(100):
            self.assertEqual(cache.get(f"key_{i}"), i)

    def test_concurrent_mixed_get_put_delete(self):
        """Test concurrent mixed operations."""
        cache = LRUCache(50)
        def worker(i):
            for j in range(50):
                cache.put(f"key_{i}", j)
                cache.get(f"key_{i}")
                if j % 5 == 0:
                    cache.delete(f"key_{i}")
        
        threads = [threading.Thread(target=worker, args=(i,)) for i in range(20)]
        for t in threads:
            t.start()
        for t in threads:
            t.join()
        
        self.assertLessEqual(cache.size(), 50)
