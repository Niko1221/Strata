class LRUCache:
    """
    A thread-safe, bounded Least Recently Used (LRU) cache implementation.
    
    This cache evicts the least recently used items when the capacity is exceeded.
    It uses an OrderedDict to maintain the order of access and a lock for thread safety.
    
    Attributes:
        capacity (int): The maximum number of items the cache can hold.
        _cache (OrderedDict): The internal storage for cache items.
        _lock (threading.Lock): A lock to ensure thread-safe operations.
    """
    
    def __init__(self, capacity: int):
        """
        Initialize the LRU Cache with a given capacity.
        
        Args:
            capacity (int): The maximum number of items to store in the cache.
            
        Raises:
            ValueError: If capacity is less than or equal to 0.
        """
        if capacity <= 0:
            raise ValueError("Capacity must be greater than 0")
        self.capacity = capacity
        self._cache = OrderedDict()
        self._lock = threading.Lock()

    def get(self, key: Any) -> Optional[Any]:
        """
        Retrieve an item from the cache.
        
        If the key exists, it is moved to the end of the OrderedDict to mark it as recently used.
        
        Args:
            key (Any): The key to look up.
            
        Returns:
            Optional[Any]: The value associated with the key, or None if not found.
        """
        with self._lock:
            if key in self._cache:
                # Move to end to mark as recently used
                self._cache.move_to_end(key)
                return self._cache[key]
            return None

    def put(self, key: Any, value: Any) -> None:
        """
        Insert or update an item in the cache.
        
        If the key already exists, its value is updated and it is moved to the end.
        If the cache is full, the least recently used item is evicted.
        
        Args:
            key (Any): The key to insert or update.
            value (Any): The value to associate with the key.
        """
        with self._lock:
            if key in self._cache:
                # Update value and move to end
                self._cache[key] = value
                self._cache.move_to_end(key)
            else:
                # Check capacity
                if len(self._cache) >= self.capacity:
                    # Evict least recently used (first item)
                    self._cache.popitem(last=False)
                # Insert new item
                self._cache[key] = value

    def delete(self, key: Any) -> bool:
        """
        Remove an item from the cache.
        
        Args:
            key (Any): The key to remove.
            
        Returns:
            bool: True if the key was found and removed, False otherwise.
        """
        with self._lock:
            if key in self._cache:
                del self._cache[key]
                return True
            return False

    def clear(self) -> None:
        """
        Remove all items from the cache.
        """
        with self._lock:
            self._cache.clear()

    def size(self) -> int:
        """
        Get the current number of items in the cache.
        
        Returns:
            int: The number of items currently stored.
        """
        with self._lock:
            return len(self._cache)

    def keys(self) -> List[Any]:
        """
        Get a list of all keys in the cache, ordered from least recently used to most recently used.
        
        Returns:
            List[Any]: A list of keys.
        """
        with self._lock:
            return list(self._cache.keys())

    def values(self) -> List[Any]:
        """
        Get a list of all values in the cache, ordered from least recently used to most recently used.
        
        Returns:
            List[Any]: A list of values.
        """
        with self._lock:
            return list(self._cache.values())

    def items(self) -> List[Tuple[Any, Any]]:
        """
        Get a list of all key-value pairs in the cache, ordered from least recently used to most recently used.
        
        Returns:
            List[Tuple[Any, Any]]: A list of (key, value) tuples.
        """
        with self._lock:
            return list(self._cache.items())

    def __contains__(self, key: Any) -> bool:
        """
        Check if a key exists in the cache.
        
        Args:
            key (Any): The key to check.
            
        Returns:
            bool: True if the key exists, False otherwise.
        """
        with self._lock:
            return key in self._cache

    def __len__(self) -> int:
        """
        Return the number of items in the cache.
        
        Returns:
            int: The number of items.
        """
        return self.size()

    def __repr__(self) -> str:
        """
        Return a string representation of the cache.
        
        Returns:
            str: A string showing the cache contents.
        """
        with self._lock:
            return f"LRUCache(capacity={self.capacity}, items={list(self._cache.items())})"

        """Test that the least recently used item is evicted when capacity is exceeded."""
        self.cache.put("a", 1)
        self.cache.put("b", 2)
        self.cache.put("c", 3)
        # Cache is full. Adding 'd' should evict 'a' (LRU)
        self.cache.put("d", 4)
        
        self.assertIsNone(self.cache.get("a"))
        self.assertEqual(self.cache.get("b"), 2)
        self.assertEqual(self.cache.get("c"), 3)
        self.assertEqual(self.cache.get("d"), 4)
        self.assertEqual(self.cache.size(), 3)

        """Test that getting an item updates its position in LRU order."""
        self.cache.put("a", 1)

        self.cache.put("c", 3)
        
        # Access 'a' to make it recently used
        self.cache.get("a")
        
        # Add 'd', should evict 'b' (now LRU)
        self.cache.put("d", 4)
        
        self.assertIsNotNone(self.cache.get("a"))
        self.assertIsNone(self.cache.get("b"))
        self.assertEqual(self.cache.get("c"), 3)
        self.assertEqual(self.cache.get("d"), 4)

        """Test concurrent mixed operations on the same key."""
        cache = LRUCache(10)
        errors = []
        
        def worker(op_type, val):
            try:
                if op_type == "put":
                    cache.put("key", val)
                elif op_type == "get":
                    cache.get("key")
                elif op_type == "delete":
                    cache.delete("key")
            except Exception as e:
                errors.append(e)

        threads = []
        for i in range(30):
            op = ["put", "get", "delete"][i % 3]
            threads.append(threading.Thread(target=worker, args=(op, i)))
        
        for t in threads:
            t.start()
        for t in threads:
            t.join()
            
        self.assertEqual(len(errors), 0)
