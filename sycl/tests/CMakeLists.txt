cmake_minimum_required(VERSION 3.16)
project(strata_sycl_host_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
enable_testing()
foreach(name wait_timeout_registry expert_cache_reverse)
    add_executable(${name}_test ${name}_test.cpp)
    target_include_directories(${name}_test PRIVATE ../include ../../include)
    target_link_libraries(${name}_test PRIVATE Threads::Threads)
    if(MSVC)
        target_compile_options(${name}_test PRIVATE /W4 /UNDEBUG)
    else()
        target_compile_options(${name}_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    endif()
    add_test(NAME ${name} COMMAND ${name}_test)
endforeach()
